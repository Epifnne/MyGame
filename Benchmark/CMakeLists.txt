add_executable(PhysicsBenchmark
	PhysicsBenchmark.cpp
)

target_link_libraries(PhysicsBenchmark
	PRIVATE
		RuntimeLib
		nlohmann_json::nlohmann_json
)

execute_process(
	COMMAND git rev-parse --short=12 HEAD
	WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
	OUTPUT_VARIABLE MYGAME_GIT_COMMIT
	OUTPUT_STRIP_TRAILING_WHITESPACE
	ERROR_QUIET
)
if(NOT MYGAME_GIT_COMMIT)
	set(MYGAME_GIT_COMMIT "unknown")
endif()
execute_process(
	COMMAND git status --porcelain
	WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
	OUTPUT_VARIABLE MYGAME_GIT_STATUS
	OUTPUT_STRIP_TRAILING_WHITESPACE
	ERROR_QUIET
)
if(MYGAME_GIT_STATUS)
	set(MYGAME_GIT_DIRTY 1)
else()
	set(MYGAME_GIT_DIRTY 0)
endif()

target_compile_definitions(PhysicsBenchmark PRIVATE
	MYGAME_PROJECT_VERSION="${PROJECT_VERSION}"
	MYGAME_COMPILER_ID="${CMAKE_CXX_COMPILER_ID}"
	MYGAME_COMPILER_VERSION="${CMAKE_CXX_COMPILER_VERSION}"
	MYGAME_BUILD_TYPE="${CMAKE_BUILD_TYPE}"
	MYGAME_GIT_COMMIT="${MYGAME_GIT_COMMIT}"
	MYGAME_GIT_DIRTY=${MYGAME_GIT_DIRTY}
)

target_compile_features(PhysicsBenchmark PRIVATE cxx_std_20)

set_target_properties(PhysicsBenchmark PROPERTIES
	RUNTIME_OUTPUT_DIRECTORY ${MYGAME_OUTPUT_ROOT}
	RUNTIME_OUTPUT_DIRECTORY_DEBUG ${MYGAME_OUTPUT_ROOT}
	RUNTIME_OUTPUT_DIRECTORY_RELEASE ${MYGAME_OUTPUT_ROOT}
	RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO ${MYGAME_OUTPUT_ROOT}
	RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL ${MYGAME_OUTPUT_ROOT}
)
